def fibonacci(n):
    # Initialize variables
    a, b = 0, 1
    result = []
    
    # While loop to generate sequence
    while a <= n:
        result.append(a)
        # Update variables
        a, b = b, a + b
        
    return result

# Test the function
n_terms = 100
print(f"Fibonacci sequence up to {n_terms}:")
print(fibonacci(n_terms))
